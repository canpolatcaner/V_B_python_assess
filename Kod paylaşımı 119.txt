import turtle as tt
import random as r


tt.speed(10)
renkler = ["red","green","blue"]


for aa in range(10):
    # tt.color(r[2])
    tt.color(r.choice(renkler))
    tt.penup()
    tt.goto(r.randint(-200,200),r.randint(-100,300))
    tt.pendown()
    kenaru = r.randint(50,150)
    for a in range(4):
        tt.forward(kenaru)
        tt.right(90)
